## 1. 模块概述

### 1.1 功能描述
控制策略模块负责储能系统的调度模式、充放电策略、保护参数和响应参数的配置与管理，为仿真计算提供控制逻辑。**本模块集成教学辅助功能，帮助学员理解不同调度策略的原理和应用场景。**

### 1.2 模块职责
- 定义调度模式（峰谷套利、负荷跟踪等）
- 定义充放电策略
- 定义保护参数和响应参数
- 生成充放电调度计划
- 提供策略校验逻辑
- 生成Gradio UI组件
- **提供策略教学说明**
- **展示策略执行过程和原理**

### 1.3 依赖关系
```
control_params.py
    ├── config/default_params.py    # 默认参数配置
    ├── models/control.py           # 控制策略数据模型
    ├── modules/basic_params.py     # BESS参数（SOC限制等）
    ├── modules/economic_params.py  # 电价时段信息
    └── modules/learning_center.py  # 教学知识关联
```

```python
@dataclass
class SchedulePoint:
    """调度计划时间点"""
    timestamp: int                      # 时间索引
    power: float                        # 功率 (kW)，正值放电，负值充电
    action: str                         # 动作描述
    soc_expected: float                 # 预期SOC


@dataclass
class DispatchSchedule:
    """调度计划"""

    schedule: List[SchedulePoint] = field(default_factory=list)
    timestep_minutes: int = 15
    total_charge_kwh: float = 0.0
    total_discharge_kwh: float = 0.0

    def get_power_array(self) -> List[float]:
        """获取功率序列"""
        return [p.power for p in self.schedule]

    def get_charge_array(self) -> List[float]:
        """获取充电功率序列（正值）"""
        return [max(0, -p.power) for p in self.schedule]

    def get_discharge_array(self) -> List[float]:
        """获取放电功率序列（正值）"""
        return [max(0, p.power) for p in self.schedule]
```

## 4. 核心逻辑

### 4.1 调度计划生成

```python
def generate_dispatch_schedule(
    control_params: ControlParams,
    bess_params,
    price_params,
    load_curve: Optional[List[float]] = None,
    timestep_minutes: int = 15,
    duration_hours: int = 24
) -> DispatchSchedule:
    """调度计划生成主函数"""

    mode = control_params.dispatch_mode

    if mode == DispatchMode.ARBITRAGE:
        return _generate_arbitrage_schedule(
            control_params, bess_params, price_params,
            timestep_minutes, duration_hours
        )
    elif mode == DispatchMode.LOAD_FOLLOW:
        if load_curve is None:
            raise ValueError("负荷跟踪模式需要提供负荷曲线")
        return _generate_load_follow_schedule(
            control_params, bess_params, price_params, load_curve,
            timestep_minutes, duration_hours
        )
    # 🔲 扩展模式预留
    elif mode == DispatchMode.RE_SMOOTHING:
        raise NotImplementedError("新能源消纳模式待实现")
    elif mode == DispatchMode.DEMAND_MANAGEMENT:
        raise NotImplementedError("需量管理模式待实现")
    else:
        raise ValueError(f"不支持的调度模式: {mode}")


def _generate_arbitrage_schedule(
    control_params: ControlParams,
    bess_params,
    price_params,
    timestep_minutes: int,
    duration_hours: int
) -> DispatchSchedule:
    """
    峰谷套利调度计划生成

    策略：谷时充电，峰时放电
    """
    from datetime import time

    schedule = []
    timesteps = int(duration_hours * 60 / timestep_minutes)

    # 获取参数
    capacity = bess_params.capacity
    power_charge = bess_params.power_charge_max
    power_discharge = bess_params.power_discharge_max
    soc_min = bess_params.soc_min
    soc_max = bess_params.soc_max
    soc_current = bess_params.soc_init
    eta_c = bess_params.efficiency_charge / 100
    eta_d = bess_params.efficiency_discharge / 100

    # 保护参数
    protect_high = control_params.protection.protect_soc_high
    protect_low = control_params.protection.protect_soc_low
    power_factor = control_params.protection.power_limit_factor

    total_charge = 0.0
    total_discharge = 0.0

    for i in range(timesteps):
        # 计算当前时间
        minutes = i * timestep_minutes
        current_time = time(minutes // 60, minutes % 60)

        # 获取当前时段电价类型
        price = price_params.get_price_at(current_time)
        period_type = _get_period_type(price, price_params)

        # 确定充放电动作
        power = 0.0
        action = "待机"

        if period_type == "valley":
            # 谷时充电
            if soc_current < protect_high:
                # 计算可充电量
                available_capacity = (protect_high - soc_current) / 100 * capacity
                max_charge_energy = power_charge * power_factor * timestep_minutes / 60
                charge_energy = min(available_capacity / eta_c, max_charge_energy)

                power = -charge_energy / (timestep_minutes / 60)  # 负值表示充电
                soc_delta = charge_energy * eta_c / capacity * 100
                soc_current += soc_delta
                total_charge += charge_energy
                action = "谷时充电"

        elif period_type == "peak":
            # 峰时放电
            if soc_current > protect_low:
                # 计算可放电量
                available_energy = (soc_current - protect_low) / 100 * capacity
                max_discharge_energy = power_discharge * power_factor * timestep_minutes / 60
                discharge_energy = min(available_energy, max_discharge_energy)

                power = discharge_energy / (timestep_minutes / 60)  # 正值表示放电
                soc_delta = discharge_energy / capacity * 100
                soc_current -= soc_delta
                total_discharge += discharge_energy * eta_d
                action = "峰时放电"

        # 平时待机
        else:
            power = 0.0
            action = "平时待机"

        schedule.append(SchedulePoint(
            timestamp=i,
            power=power,
            action=action,
            soc_expected=soc_current
        ))

    return DispatchSchedule(
        schedule=schedule,
        timestep_minutes=timestep_minutes,
        total_charge_kwh=total_charge,
        total_discharge_kwh=total_discharge
    )


def _generate_load_follow_schedule(
    control_params: ControlParams,
    bess_params,
    price_params,
    load_curve: List[float],
    timestep_minutes: int,
    duration_hours: int
) -> DispatchSchedule:
    """
    负荷跟踪调度计划生成

    策略：削峰填谷，平滑负荷曲线
    """
    schedule = []
    timesteps = len(load_curve)

    # 计算目标平均负荷
    avg_load = sum(load_curve) / len(load_curve)

    # 获取参数
    capacity = bess_params.capacity
    power_charge = bess_params.power_charge_max
    power_discharge = bess_params.power_discharge_max
    soc_current = bess_params.soc_init
    eta_c = bess_params.efficiency_charge / 100
    eta_d = bess_params.efficiency_discharge / 100

    protect_high = control_params.protection.protect_soc_high
    protect_low = control_params.protection.protect_soc_low

    total_charge = 0.0
    total_discharge = 0.0

    for i, load in enumerate(load_curve):
        power = 0.0
        action = "待机"

        deviation = load - avg_load

        if deviation > 0:
            # 负荷高于平均，放电削峰
            if soc_current > protect_low:
                discharge_power = min(deviation, power_discharge)
                available_energy = (soc_current - protect_low) / 100 * capacity
                max_discharge = available_energy / (timestep_minutes / 60)

                power = min(discharge_power, max_discharge)
                soc_delta = power * (timestep_minutes / 60) / capacity * 100
                soc_current -= soc_delta
                total_discharge += power * (timestep_minutes / 60) * eta_d
                action = "削峰放电"

        elif deviation < 0:
            # 负荷低于平均，充电填谷
            if soc_current < protect_high:
                charge_power = min(-deviation, power_charge)
                available_capacity = (protect_high - soc_current) / 100 * capacity
                max_charge = available_capacity / eta_c / (timestep_minutes / 60)

                power = -min(charge_power, max_charge)  # 负值表示充电
                soc_delta = -power * (timestep_minutes / 60) * eta_c / capacity * 100
                soc_current += soc_delta
                total_charge += -power * (timestep_minutes / 60)
                action = "填谷充电"

        schedule.append(SchedulePoint(
            timestamp=i,
            power=power,
            action=action,
            soc_expected=soc_current
        ))

    return DispatchSchedule(
        schedule=schedule,
        timestep_minutes=timestep_minutes,
        total_charge_kwh=total_charge,
        total_discharge_kwh=total_discharge
    )


def _get_period_type(price: float, price_params) -> str:
    """根据电价判断时段类型"""
    if price <= price_params.price_valley:
        return "valley"
    elif price >= price_params.price_peak:
        return "peak"
    else:
        return "flat"
```

### 4.2 参数校验逻辑

```python
def validate_control_params(
    params: ControlParams,
    bess_params
) -> tuple[bool, list[str]]:
    """控制参数校验"""
    errors = []

    # 1. 保护阈值校验
    protection = params.protection

    # 过充保护必须大于SOC上限
    if protection.protect_soc_high < bess_params.soc_max:
        errors.append(
            f"过充保护阈值({protection.protect_soc_high}%)应大于等于SOC上限({bess_params.soc_max}%)"
        )

    # 过放保护必须小于SOC下限
    if protection.protect_soc_low > bess_params.soc_min:
        errors.append(
            f"过放保护阈值({protection.protect_soc_low}%)应小于等于SOC下限({bess_params.soc_min}%)"
        )

    # 保护阈值合理性
    if protection.protect_soc_high <= protection.protect_soc_low:
        errors.append("过充保护阈值必须大于过放保护阈值")

    # 2. 功率限制系数校验
    if not (0.5 <= protection.power_limit_factor <= 1.5):
        errors.append("功率限制系数应在0.5-1.5范围内")

    # 3. 策略参数校验
    strategy = params.strategy

    # 价格触发策略需要阈值
    if strategy.charge_strategy == ChargeStrategy.PRICE_TRIGGER_LOW:
        if strategy.price_charge_threshold is None:
            errors.append("价格触发充电策略需要设置电价阈值")

    if strategy.discharge_strategy == DischargeStrategy.PRICE_TRIGGER_HIGH:
        if strategy.price_discharge_threshold is None:
            errors.append("价格触发放电策略需要设置电价阈值")

    # SOC触发策略需要阈值
    if strategy.charge_strategy == ChargeStrategy.SOC_TRIGGER_LOW:
        if strategy.soc_charge_trigger is None:
            errors.append("SOC触发充电策略需要设置SOC阈值")

    if strategy.discharge_strategy == DischargeStrategy.SOC_TRIGGER_HIGH:
        if strategy.soc_discharge_trigger is None:
            errors.append("SOC触发放电策略需要设置SOC阈值")

    # 4. 响应参数校验
    response = params.response
    if response.response_time < 10:
        errors.append("响应时间不能小于10ms")

    if response.ramp_rate <= 0:
        errors.append("爬坡速率必须大于0")

    # 🔲 扩展校验预留

    return len(errors) == 0, errors
```

```python
def setup_control_params_events(components: dict):
    """设置控制策略模块的事件绑定"""

    # 调度模式变更 -> 更新描述
    mode_descriptions = {
        "ARBITRAGE": "_峰谷套利：在谷电时段充电，峰电时段放电，赚取电价差收益_",
        "LOAD_FOLLOW": "_负荷跟踪：根据负荷曲线削峰填谷，平滑用电负荷_",
        "RE_SMOOTHING": "_新能源消纳：配合光伏/风电出力，平滑波动（待实现）_",
        "DEMAND_MANAGEMENT": "_需量管理：控制最大需量，降低容量电费（待实现）_",
    }

    components["dispatch_mode"].change(
        fn=lambda m: mode_descriptions.get(m, ""),
        inputs=[components["dispatch_mode"]],
        outputs=[components["mode_description"]]
    )

    # 充放电策略变更 -> 显示/隐藏触发阈值
    def update_trigger_visibility(charge_s, discharge_s):
        show = (
            charge_s == "PRICE_TRIGGER_LOW" or
            discharge_s == "PRICE_TRIGGER_HIGH"
        )
        return gr.Row(visible=show)

    components["charge_strategy"].change(
        fn=update_trigger_visibility,
        inputs=[components["charge_strategy"], components["discharge_strategy"]],
        outputs=[components["trigger_params_row"]]
    )

    components["discharge_strategy"].change(
        fn=update_trigger_visibility,
        inputs=[components["charge_strategy"], components["discharge_strategy"]],
        outputs=[components["trigger_params_row"]]
    )
```

```python
# 🔲 自定义调度器注册机制（预留）
class SchedulerRegistry:
    """调度器注册表"""

    _schedulers: Dict[str, Callable] = {}

    @classmethod
    def register(cls, name: str, scheduler: Callable):
        """注册自定义调度器"""
        cls._schedulers[name] = scheduler

    @classmethod
    def get(cls, name: str) -> Optional[Callable]:
        """获取调度器"""
        return cls._schedulers.get(name)

    @classmethod
    def list_all(cls) -> List[str]:
        """列出所有已注册的调度器"""
        return list(cls._schedulers.keys())
```
