## 1. 模块概述

### 1.1 功能描述
教学引导模块是平台的学习入口和知识管理中心，提供知识地图导航、分步教程、案例库、术语表等功能，帮助学员系统性地学习储能EMS知识。

### 1.2 模块职责
- 展示知识体系结构（知识地图）
- 追踪学员学习进度
- 提供分步骤教程引导
- 管理典型案例库
- 提供术语快速查询
- 推荐学习内容
- 关联知识点到各功能模块

### 1.3 依赖关系
```
learning_center.py
    ├── data/knowledge_base/        # 知识库数据
    │   ├── concepts.json           # 概念定义
    │   ├── terminology.json        # 术语表
    │   └── learning_paths.json     # 学习路径
    ├── data/case_library/          # 案例库
    ├── config/teaching_content.py  # 教学配置
    └── models/teaching.py          # 教学数据模型
```

```python
from dataclasses import dataclass, field
from typing import Optional, List, Dict, Any
from enum import Enum
from datetime import datetime


class KnowledgeLevel(Enum):
    """知识点难度等级"""
    BEGINNER = "入门"
    INTERMEDIATE = "进阶"
    ADVANCED = "高级"


class ContentType(Enum):
    """内容类型"""
    TEXT = "文本"
    IMAGE = "图片"
    VIDEO = "视频"
    ANIMATION = "动画"
    INTERACTIVE = "交互"


@dataclass
class KnowledgePoint:
    """知识点数据类"""

    id: str                             # 唯一标识，如 "1.2.3"
    title: str                          # 标题
    parent_id: Optional[str] = None     # 父节点ID
    level: KnowledgeLevel = KnowledgeLevel.BEGINNER

    # 内容
    summary: str = ""                   # 简要概述
    content: str = ""                   # 详细内容（Markdown格式）
    content_type: ContentType = ContentType.TEXT
    media_url: Optional[str] = None     # 媒体资源URL

    # 关联
    related_points: List[str] = field(default_factory=list)  # 关联知识点
    related_params: List[str] = field(default_factory=list)  # 关联参数
    related_modules: List[str] = field(default_factory=list) # 关联功能模块

    # 教学
    learning_time_minutes: int = 5      # 预计学习时间
    keywords: List[str] = field(default_factory=list)

    # 🔲 扩展预留
    ext_data: Dict[str, Any] = field(default_factory=dict)


@dataclass
class KnowledgeTree:
    """知识树结构"""

    root_nodes: List[str] = field(default_factory=list)  # 根节点ID列表
    nodes: Dict[str, KnowledgePoint] = field(default_factory=dict)

    def get_node(self, node_id: str) -> Optional[KnowledgePoint]:
        """获取节点"""
        return self.nodes.get(node_id)

    def get_children(self, node_id: str) -> List[KnowledgePoint]:
        """获取子节点列表"""
        return [
            node for node in self.nodes.values()
            if node.parent_id == node_id
        ]

    def get_path(self, node_id: str) -> List[str]:
        """获取从根到该节点的路径"""
        path = []
        current = node_id
        while current:
            path.insert(0, current)
            node = self.nodes.get(current)
            current = node.parent_id if node else None
        return path
```

```python
@dataclass
class LearningProgress:
    """学习进度"""

    user_id: str

    started_at: Optional[datetime] = None

@dataclass
class UserLearningState:
    """用户学习状态汇总"""

    user_id: str
    total_points: int = 0               # 总知识点数
    completed_points: int = 0           # 已完成数
    in_progress_points: int = 0         # 进行中数
    total_time_minutes: int = 0         # 总学习时长
    current_path: str = ""              # 当前学习路径
    current_point: str = ""             # 当前学习节点
    weak_areas: List[str] = field(default_factory=list)  # 薄弱领域
    recommended_next: List[str] = field(default_factory=list)  # 推荐学习

    @property
    def completion_percentage(self) -> float:
        """计算完成百分比"""
        if self.total_points == 0:
            return 0.0
        return self.completed_points / self.total_points * 100
```

```python
@dataclass
class Terminology:
    """术语"""

    term: str                           # 术语
    english: str = ""                   # 英文
    abbreviation: str = ""              # 缩写
    definition: str = ""                # 定义
    explanation: str = ""               # 详细解释
    related_terms: List[str] = field(default_factory=list)
    related_points: List[str] = field(default_factory=list)
    example: str = ""                   # 使用示例


@dataclass
class TerminologyDict:
    """术语表"""

    terms: Dict[str, Terminology] = field(default_factory=dict)

    def search(self, query: str) -> List[Terminology]:
        """搜索术语"""
        query_lower = query.lower()
        results = []
        for term in self.terms.values():
            if (query_lower in term.term.lower() or
                query_lower in term.english.lower() or
                query_lower in term.abbreviation.lower()):
                results.append(term)
        return results
```

### 4.2 学习进度追踪

```python
class LearningProgressTracker:
    """学习进度追踪器"""

    def __init__(self, storage_path: str = "data/user_progress"):
        self.storage_path = storage_path
        self._progress_cache: Dict[str, Dict[str, LearningProgress]] = {}

    def get_user_state(self, user_id: str) -> UserLearningState:
        """获取用户学习状态"""
        progress_dict = self._load_progress(user_id)
        tree = get_knowledge_tree()

        state = UserLearningState(user_id=user_id)
        state.total_points = len(tree.nodes)

        for point_id, progress in progress_dict.items():
            if progress.status == "completed":
                state.completed_points += 1
            elif progress.status == "in_progress":
                state.in_progress_points += 1
                state.current_point = point_id

            state.total_time_minutes += progress.time_spent_seconds // 60

        # 分析薄弱领域
        state.weak_areas = self._analyze_weak_areas(user_id, progress_dict)

        # 生成推荐内容
        state.recommended_next = self._generate_recommendations(
            user_id, progress_dict, tree
        )

        return state

    def update_progress(
        self,
        user_id: str,
        point_id: str,
        status: str,
        time_spent: int = 0
    ) -> bool:
        """更新学习进度"""
        progress_dict = self._load_progress(user_id)

        if point_id not in progress_dict:
            progress_dict[point_id] = LearningProgress(
                user_id=user_id,
                knowledge_point_id=point_id,
                started_at=datetime.now()
            )

        progress = progress_dict[point_id]
        progress.status = status
        progress.time_spent_seconds += time_spent

        if status == "completed" and progress.completed_at is None:
            progress.completed_at = datetime.now()

        self._save_progress(user_id, progress_dict)
        return True

    def _analyze_weak_areas(
        self,
        user_id: str,
        progress_dict: Dict[str, LearningProgress]
    ) -> List[str]:
        """分析薄弱领域"""
        weak = []

        for point_id, progress in progress_dict.items():
            # 测验得分低的
            if progress.quiz_score is not None and progress.quiz_score < 60:
                weak.append(point_id)
            # 学习时间过短的（可能没认真学）
            point = get_knowledge_point(point_id)
            if point and progress.status == "completed":
                expected = point.learning_time_minutes * 60
                if progress.time_spent_seconds < expected * 0.3:
                    weak.append(point_id)

        return list(set(weak))

    def _generate_recommendations(
        self,
        user_id: str,
        progress_dict: Dict[str, LearningProgress],
        tree: KnowledgeTree
    ) -> List[str]:
        """生成学习推荐"""
        recommendations = []

        # 1. 未完成的知识点
        for point_id in tree.nodes:
            if point_id not in progress_dict:
                # 检查前置知识是否完成
                parent = tree.nodes[point_id].parent_id
                if parent is None or (
                    parent in progress_dict and
                    progress_dict[parent].status == "completed"
                ):
                    recommendations.append(point_id)
                    if len(recommendations) >= 3:
                        break

        # 2. 薄弱领域复习
        weak = self._analyze_weak_areas(user_id, progress_dict)
        for point_id in weak[:2]:
            if point_id not in recommendations:
                recommendations.append(point_id)

        return recommendations[:5]
```

```python
def _render_knowledge_tree_html() -> str:
    """渲染知识树HTML"""
    tree = get_knowledge_tree()

    html = '<div class="knowledge-tree">'
    for root_id in tree.root_nodes:
        html += _render_tree_node(tree, root_id, 0)
    html += '</div>'

    return html


def _render_tree_node(tree: KnowledgeTree, node_id: str, level: int) -> str:
    """递归渲染树节点"""
    node = tree.get_node(node_id)
    if not node:
        return ""

    indent = "  " * level
    status_icon = "○"  # 未完成
    # status_icon = "◉" # 进行中
    # status_icon = "✓" # 已完成

    html = f'{indent}<div class="tree-node level-{level}">'
    html += f'<span class="status">{status_icon}</span>'
    html += f'<span class="title" data-id="{node_id}">{node.title}</span>'

    children = tree.get_children(node_id)
    if children:
        html += '<div class="children">'
        for child in children:
            html += _render_tree_node(tree, child.id, level + 1)
        html += '</div>'

    html += '</div>'
    return html
```
