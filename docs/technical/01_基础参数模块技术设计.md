## 1. 模块概述

### 1.1 功能描述
基础参数模块负责储能设备物理参数的配置、校验、存储和UI展示。当前版本支持电池储能系统(BESS)，预留其他储能类型扩展接口。**本模块集成教学辅助功能，为每个参数提供概念解释、影响分析和知识关联。**

### 1.2 模块职责
- 定义储能设备参数数据结构
- 提供参数校验逻辑
- 生成Gradio UI组件
- 参数的导入/导出
- 参数变更事件处理
- **提供参数教学提示（概念解释、影响分析）**
- **关联知识点和学习资源**

### 1.3 依赖关系
```
basic_params.py
    ├── config/default_params.py    # 默认参数配置
    ├── models/bess.py              # BESS数据模型
    ├── utils/validation.py         # 参数校验工具（预留）
    └── modules/learning_center.py  # 教学知识关联
```

```python
@dataclass
class ParamConstraint:
    """参数约束定义"""
    min_value: Optional[float] = None
    max_value: Optional[float] = None
    required: bool = True
    depends_on: Optional[str] = None      # 依赖其他参数
    validator: Optional[callable] = None  # 自定义校验函数


# BESS参数约束配置
BESS_CONSTRAINTS: Dict[str, ParamConstraint] = {
    "capacity": ParamConstraint(min_value=10, max_value=10000, required=True),
    "power_rated": ParamConstraint(min_value=5, max_value=5000, required=True),
    "power_charge_max": ParamConstraint(
        min_value=0,
        depends_on="power_rated",  # 不能超过额定功率
```

## 4. 核心逻辑

### 4.1 参数校验流程

```python
def validate_bess_params(params: BESSParams) -> tuple[bool, list[str]]:
    """
    BESS参数校验完整流程
    """
    errors = []

    # 1. 基础范围校验
    for param_name, constraint in BESS_CONSTRAINTS.items():
        value = getattr(params, param_name, None)

        # 必填校验
        if constraint.required and value is None:
            errors.append(f"{param_name} 是必填参数")
            continue

        if value is None:
            continue

        # 范围校验
        if constraint.min_value is not None and value < constraint.min_value:
            errors.append(f"{param_name} 不能小于 {constraint.min_value}")
        if constraint.max_value is not None and value > constraint.max_value:
            errors.append(f"{param_name} 不能大于 {constraint.max_value}")

    # 2. 依赖关系校验
    # 充电功率不能超过额定功率
    if params.power_charge_max > params.power_rated:
        errors.append("充电功率上限不能超过额定功率")

    # 放电功率不能超过额定功率
    if params.power_discharge_max > params.power_rated:
        errors.append("放电功率上限不能超过额定功率")

    # SOC初始值必须在范围内
    if not (params.soc_min <= params.soc_init <= params.soc_max):
        errors.append("初始SOC必须在SOC上下限范围内")

    # SOC目标值必须在范围内
    if params.soc_target is not None:
        if not (params.soc_min <= params.soc_target <= params.soc_max):
            errors.append("目标SOC必须在SOC上下限范围内")

    # 3. 逻辑合理性校验
    # SOC上限必须大于下限
    if params.soc_max <= params.soc_min:
        errors.append("SOC上限必须大于SOC下限")

    # 容量功率比合理性（一般在0.5-4小时之间）
    duration = params.capacity / params.power_rated
    if duration < 0.25 or duration > 8:
        errors.append(f"容量/功率比({duration:.1f}h)异常，建议在0.5-4小时范围")

    # 4. 自定义校验器（如有）
    for param_name, constraint in BESS_CONSTRAINTS.items():
        if constraint.validator:
            value = getattr(params, param_name, None)
            is_valid, msg = constraint.validator(value, params)
            if not is_valid:
                errors.append(msg)

    # 🔲 扩展校验预留
    # errors.extend(_ext_validation(params))

    return len(errors) == 0, errors
```

### 4.2 参数联动逻辑

```python
def _on_param_change(param_name: str, value: Any, current_params: dict) -> dict:
    """
    参数变更时的联动处理

    Returns:
        dict: 需要更新的其他参数 {param_name: new_value}
    """
    updates = {}

    # 额定功率变更 -> 更新充放电功率上限的最大值
    if param_name == "power_rated":
        if current_params.get("power_charge_max", 0) > value:
            updates["power_charge_max"] = value
        if current_params.get("power_discharge_max", 0) > value:
            updates["power_discharge_max"] = value

    # SOC下限变更 -> 检查初始SOC和目标SOC
    if param_name == "soc_min":
        if current_params.get("soc_init", 0) < value:
            updates["soc_init"] = value
        if current_params.get("soc_target") is not None:
            if current_params["soc_target"] < value:
                updates["soc_target"] = value

    # SOC上限变更 -> 检查初始SOC和目标SOC
    if param_name == "soc_max":
        if current_params.get("soc_init", 100) > value:
            updates["soc_init"] = value
        if current_params.get("soc_target") is not None:
            if current_params["soc_target"] > value:
                updates["soc_target"] = value

    # 🔲 扩展联动预留

    return updates
```

```python
def _create_param_with_help(
    label: str,
    param_name: str,
    component_type: str,
    **kwargs
) -> gr.Column:
    """创建带教学提示的参数组件"""

    meta = BESS_TEACHING_META.get(param_name, ParamTeachingMeta())

    with gr.Column():
        with gr.Row():
            # 参数标签
            gr.Markdown(f"**{label}**")
            # 帮助按钮
            help_btn = gr.Button("?", size="sm", scale=0)

        # 参数输入组件
        if component_type == "number":
            component = gr.Number(**kwargs)
        elif component_type == "slider":
            component = gr.Slider(**kwargs)
        elif component_type == "dropdown":
            component = gr.Dropdown(**kwargs)

        # 教学提示区域（默认隐藏）
        with gr.Accordion("参数说明", open=False, visible=False) as help_panel:
            gr.Markdown(f"""
**概念解释**: {meta.help_text}

**影响分析**: {meta.impact_desc}

**典型范围**: {meta.typical_range}

**关联知识点**: {', '.join(meta.learning_points) if meta.learning_points else '无'}
            """)
            if meta.warning_conditions:
                gr.Markdown(f"⚠️ **注意**: {'; '.join(meta.warning_conditions)}")

        # 实时影响提示（参数变更时显示）
        impact_hint = gr.Markdown("", visible=False)

    return component, help_btn, help_panel, impact_hint


def _show_param_impact(param_name: str, old_value: Any, new_value: Any) -> str:

    if not meta:
        return ""

    hint = f"📊 **{meta.impact_desc}**"

    # 检查是否触发警告条件
    for warning in meta.warning_conditions:
```

### 5.2 事件绑定

```python
def setup_basic_params_events(components: dict):
    """设置基础参数模块的事件绑定"""

    # 额定功率变更 -> 更新充放电功率上限
    components["power_rated"].change(
        fn=lambda p: (min(p, components["power_charge_max"].value),
                      min(p, components["power_discharge_max"].value)),
        inputs=[components["power_rated"]],
        outputs=[components["power_charge_max"], components["power_discharge_max"]]
    )

    # SOC范围变更 -> 更新初始SOC和目标SOC的范围
    components["soc_range"].change(
        fn=_update_soc_constraints,
        inputs=[components["soc_range"]],
        outputs=[components["soc_init"], components["soc_target"]]
    )

    # 重置按钮
    components["reset_btn"].click(
        fn=_reset_all_params,
        outputs=list(components.values())
    )

    # 保存按钮
    components["save_btn"].click(
        fn=_save_and_validate,
        inputs=[v for k, v in components.items() if k not in ["reset_btn", "save_btn", "validation_output"]],
        outputs=[components["validation_output"]]
    )
```
