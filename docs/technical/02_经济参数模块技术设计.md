## 1. 模块概述

### 1.1 功能描述
经济参数模块负责电价策略、成本参数、财务参数的配置与管理，为储能系统经济性分析提供数据支持。**本模块集成教学辅助功能，帮助学员理解电价策略、成本构成和财务指标计算方法。**

### 1.2 模块职责
- 定义电价参数和时段划分
- 定义投资成本和运营成本参数
- 定义财务分析参数
- 提供参数校验逻辑
- 生成Gradio UI组件
- 支持多电价方案配置（预留）
- **提供经济参数教学提示**
- **关联经济分析知识点**

```python
from dataclasses import dataclass, field
from typing import Optional, List, Dict, Any
from enum import Enum
from datetime import time


class PricePeriodType(Enum):
    """电价时段类型"""
    VALLEY = "谷时"
    FLAT = "平时"
    PEAK = "峰时"
    CRITICAL = "尖峰"
    # 🔲 扩展预留
    EXT_PERIOD_1 = "_扩展时段1"
    EXT_PERIOD_2 = "_扩展时段2"


@dataclass
class TimePeriod:
    """时段定义"""
    start: time          # 开始时间
    end: time            # 结束时间
    period_type: PricePeriodType

    def contains(self, t: time) -> bool:
        """判断时间点是否在此时段内"""
        if self.start <= self.end:
            return self.start <= t < self.end
        else:  # 跨日时段
            return t >= self.start or t < self.end

    def duration_hours(self) -> float:
        """计算时段时长（小时）"""
        start_minutes = self.start.hour * 60 + self.start.minute
        end_minutes = self.end.hour * 60 + self.end.minute
        if end_minutes > start_minutes:
            return (end_minutes - start_minutes) / 60
        else:  # 跨日
            return (24 * 60 - start_minutes + end_minutes) / 60


@dataclass
class ElectricityPriceParams:
    """电价参数数据类"""

    # === 基础电价 (元/kWh) ===
    price_valley: float = 0.4           # 谷时电价
    price_flat: float = 0.8             # 平时电价
    price_peak: float = 1.2             # 峰时电价
    price_critical: Optional[float] = None  # 尖峰电价（可选）
    price_feed_in: Optional[float] = 0.5    # 上网电价
    # 🔲 扩展预留
    ext_price_1: Optional[float] = None
    ext_price_2: Optional[float] = None

    # === 时段划分 ===
    periods: List[TimePeriod] = field(default_factory=list)

    # === 元数据 ===
    scheme_name: str = "默认电价方案"
    description: str = ""

    def __post_init__(self):
        """初始化默认时段"""
        if not self.periods:
            self.periods = self._default_periods()

    def _default_periods(self) -> List[TimePeriod]:
        """默认时段配置"""
        return [
            TimePeriod(time(0, 0), time(8, 0), PricePeriodType.VALLEY),
            TimePeriod(time(8, 0), time(10, 0), PricePeriodType.FLAT),
            TimePeriod(time(10, 0), time(12, 0), PricePeriodType.PEAK),
            TimePeriod(time(12, 0), time(18, 0), PricePeriodType.FLAT),
            TimePeriod(time(18, 0), time(22, 0), PricePeriodType.PEAK),
            TimePeriod(time(22, 0), time(24, 0), PricePeriodType.FLAT),
        ]

    def get_price_at(self, t: time) -> float:
        """获取指定时间的电价"""
        for period in self.periods:
            if period.contains(t):
                price_map = {
                    PricePeriodType.VALLEY: self.price_valley,
                    PricePeriodType.FLAT: self.price_flat,
                    PricePeriodType.PEAK: self.price_peak,
                    PricePeriodType.CRITICAL: self.price_critical or self.price_peak,
                }
                return price_map.get(period.period_type, self.price_flat)
        return self.price_flat  # 默认返回平时电价

    def get_price_curve(self, timestep_minutes: int = 15) -> List[float]:
        """生成24小时电价曲线"""
        prices = []
        for i in range(0, 24 * 60, timestep_minutes):
            t = time(i // 60, i % 60)
            prices.append(self.get_price_at(t))
        return prices
```

```python
@dataclass
class CostParams:
    """成本参数数据类"""

    # === 投资成本 ===
    cost_per_kwh: float = 1500.0        # 元/kWh，单位容量成本
    cost_per_kw: float = 500.0          # 元/kW，单位功率成本
    cost_install_ratio: float = 10.0    # %，安装成本比例
    # 🔲 扩展预留
    ext_cost_1: Optional[float] = None

    # === 运营成本 ===
    cost_om_ratio: float = 2.0          # %，年运维成本比例
    cost_insurance_ratio: float = 0.5   # %，保险费率
    # 🔲 扩展预留
    ext_om_1: Optional[float] = None

    def calculate_initial_investment(
        self,
        capacity_kwh: float,
        power_kw: float
    ) -> float:
        """
        计算初始投资成本

        Returns:
            float: 总初始投资（元）
        """
        equipment_cost = (
            capacity_kwh * self.cost_per_kwh +
            power_kw * self.cost_per_kw
        )
        install_cost = equipment_cost * self.cost_install_ratio / 100
        return equipment_cost + install_cost

    def calculate_annual_opex(self, initial_investment: float) -> float:
        """
        计算年运营成本

        Returns:
            float: 年运营成本（元）
        """
        om_cost = initial_investment * self.cost_om_ratio / 100
        insurance_cost = initial_investment * self.cost_insurance_ratio / 100
        return om_cost + insurance_cost
```

```python
@dataclass
class FinancialParams:
    """财务参数数据类"""

    discount_rate: float = 8.0          # %，贴现率
    project_years: int = 15             # 年，项目周期
    residual_ratio: float = 5.0         # %，残值率
    # 🔲 扩展预留
    ext_fin_1: Optional[float] = None

    def calculate_npv(
        self,
        initial_investment: float,
        annual_cash_flows: List[float]
    ) -> float:
        """
        计算净现值 (NPV)

        Args:
            initial_investment: 初始投资
            annual_cash_flows: 各年现金流列表

        Returns:
            float: NPV值
        """
        npv = -initial_investment
        r = self.discount_rate / 100

        for year, cf in enumerate(annual_cash_flows, 1):
            npv += cf / ((1 + r) ** year)

        # 残值
        residual = initial_investment * self.residual_ratio / 100
        npv += residual / ((1 + r) ** len(annual_cash_flows))

        return npv

    def calculate_payback_period(
        self,
        initial_investment: float,
        annual_revenue: float
    ) -> float:
        """
        计算静态投资回收期

        Returns:
            float: 回收期（年）
        """
        if annual_revenue <= 0:
            return float('inf')
        return initial_investment / annual_revenue

    def calculate_dynamic_payback(
        self,
        initial_investment: float,
        annual_cash_flows: List[float]
    ) -> float:
        """
        计算动态投资回收期

        Returns:
            float: 动态回收期（年）
        """
        r = self.discount_rate / 100
        cumulative = -initial_investment

        for year, cf in enumerate(annual_cash_flows, 1):
            discounted_cf = cf / ((1 + r) ** year)
            cumulative += discounted_cf
            if cumulative >= 0:
                # 线性插值计算精确回收期
                prev_cumulative = cumulative - discounted_cf
                fraction = -prev_cumulative / discounted_cf
                return year - 1 + fraction

        return float('inf')
```

## 4. 核心逻辑

### 4.1 时段解析与校验

```python
def parse_time_periods(period_strings: Dict[str, List[str]]) -> List[TimePeriod]:
    """
    解析时段字符串配置

    Args:
        period_strings: {
            "peak": ["10:00-12:00", "18:00-22:00"],
            "flat": ["08:00-10:00", "12:00-18:00"],
            "valley": ["00:00-08:00", "22:00-24:00"]
        }

    Returns:
        List[TimePeriod]: 时段列表
    """
    periods = []
    type_map = {
        "valley": PricePeriodType.VALLEY,
        "flat": PricePeriodType.FLAT,
        "peak": PricePeriodType.PEAK,
        "critical": PricePeriodType.CRITICAL,
    }

    for period_type, time_ranges in period_strings.items():
        if period_type not in type_map:
            continue
        for time_range in time_ranges:
            start_str, end_str = time_range.split("-")
            start = time.fromisoformat(start_str.strip())
            end = time.fromisoformat(end_str.strip())
            periods.append(TimePeriod(start, end, type_map[period_type]))

    return periods


def validate_time_periods(periods: List[TimePeriod]) -> tuple[bool, list[str]]:
    """
    校验时段配置
    - 检查是否覆盖24小时
    - 检查是否有重叠

    Returns:
        tuple: (是否通过, 错误信息列表)
    """
    errors = []

    # 1. 检查总时长是否为24小时
    total_hours = sum(p.duration_hours() for p in periods)
    if abs(total_hours - 24) > 0.01:
        errors.append(f"时段总时长({total_hours:.1f}h)不等于24小时")

    # 2. 检查时段重叠
    for i, p1 in enumerate(periods):
        for p2 in periods[i+1:]:
            if _periods_overlap(p1, p2):
                errors.append(
                    f"时段重叠: {p1.start}-{p1.end} 与 {p2.start}-{p2.end}"
                )

    # 🔲 扩展校验预留

    return len(errors) == 0, errors


def _periods_overlap(p1: TimePeriod, p2: TimePeriod) -> bool:
    """检查两个时段是否重叠"""
    # 简化处理：转换为分钟区间检查
    def to_minutes(t: time) -> int:
        return t.hour * 60 + t.minute

    s1, e1 = to_minutes(p1.start), to_minutes(p1.end)
    s2, e2 = to_minutes(p2.start), to_minutes(p2.end)

    # 处理跨日情况
    if e1 == 0:
        e1 = 24 * 60
    if e2 == 0:
        e2 = 24 * 60

    return not (e1 <= s2 or e2 <= s1)
```

### 4.2 经济性计算

```python
def calculate_daily_revenue(
    price_params: ElectricityPriceParams,
    charge_schedule: List[float],    # 每时段充电功率 (kW)
    discharge_schedule: List[float], # 每时段放电功率 (kW)
    timestep_hours: float = 0.25     # 时间步长 (小时)
) -> dict:
    """
    计算日收益

    Returns:
        dict: {
            'charge_cost': float,      # 充电成本
            'discharge_revenue': float, # 放电收益
            'arbitrage_profit': float,  # 套利利润
            'details': List[dict]       # 各时段明细
        }
    """
    price_curve = price_params.get_price_curve(int(timestep_hours * 60))

    charge_cost = 0.0
    discharge_revenue = 0.0
    details = []

    for i, (charge_p, discharge_p, price) in enumerate(
        zip(charge_schedule, discharge_schedule, price_curve)
    ):
        # 充电成本
        charge_energy = charge_p * timestep_hours
        cost = charge_energy * price
        charge_cost += cost

        # 放电收益
        discharge_energy = discharge_p * timestep_hours
        revenue = discharge_energy * price
        discharge_revenue += revenue

        details.append({
            'timestep': i,
            'price': price,
            'charge_kWh': charge_energy,
            'charge_cost': cost,
            'discharge_kWh': discharge_energy,
            'discharge_revenue': revenue,
        })

    return {
        'charge_cost': charge_cost,
        'discharge_revenue': discharge_revenue,
        'arbitrage_profit': discharge_revenue - charge_cost,
        'details': details,
    }


def calculate_annual_revenue(
    daily_profit: float,
    operating_days: int = 365
) -> float:
    """计算年化收益"""
    return daily_profit * operating_days


def calculate_full_economics(
    bess_params,            # BESSParams
    economic_params,        # EconomicParams
    daily_profit: float
) -> dict:
    """
    计算完整经济指标

    Returns:
        dict: 完整经济分析结果
    """
    cost = economic_params.cost_params
    fin = economic_params.financial_params

    # 初始投资
    initial_investment = cost.calculate_initial_investment(
        bess_params.capacity,
        bess_params.power_rated
    )

    # 年运营成本
    annual_opex = cost.calculate_annual_opex(initial_investment)

    # 年收益（考虑运营成本）
    annual_revenue = calculate_annual_revenue(daily_profit)
    net_annual_revenue = annual_revenue - annual_opex

    # 投资回收期
    payback = fin.calculate_payback_period(initial_investment, net_annual_revenue)

    # 生成年度现金流
    annual_cash_flows = []
    for year in range(1, fin.project_years + 1):
        # 考虑容量衰减对收益的影响
        degradation_factor = 1 - (bess_params.degradation_rate / 100) * year
        year_revenue = net_annual_revenue * max(0.7, degradation_factor)
        annual_cash_flows.append(year_revenue)

    # NPV
    npv = fin.calculate_npv(initial_investment, annual_cash_flows)

    # 动态回收期
    dynamic_payback = fin.calculate_dynamic_payback(
        initial_investment,
        annual_cash_flows
    )

    return {
        'initial_investment': initial_investment,
        'annual_opex': annual_opex,
        'annual_revenue': annual_revenue,
        'net_annual_revenue': net_annual_revenue,
        'static_payback': payback,
        'dynamic_payback': dynamic_payback,
        'npv': npv,
        # 🔲 扩展指标预留
        'irr': None,  # 待实现
        'lcoe': None, # 待实现
    }
```

```python
import matplotlib.pyplot as plt
import numpy as np

def plot_price_curve(price_params: ElectricityPriceParams) -> plt.Figure:
    """绘制24小时电价曲线"""
    fig, ax = plt.subplots(figsize=(10, 4))

    prices = price_params.get_price_curve(15)  # 15分钟间隔
    times = np.arange(0, 24, 0.25)

    # 绘制阶梯图
    ax.step(times, prices, where='post', linewidth=2, color='#2196F3')
    ax.fill_between(times, prices, step='post', alpha=0.3, color='#2196F3')

    # 添加电价水平线
    ax.axhline(y=price_params.price_valley, color='green',
               linestyle='--', alpha=0.5, label=f'谷时 {price_params.price_valley}元')
    ax.axhline(y=price_params.price_flat, color='orange',
               linestyle='--', alpha=0.5, label=f'平时 {price_params.price_flat}元')
    ax.axhline(y=price_params.price_peak, color='red',
               linestyle='--', alpha=0.5, label=f'峰时 {price_params.price_peak}元')

    ax.set_xlabel('时间 (h)')
    ax.set_ylabel('电价 (元/kWh)')
    ax.set_title('24小时电价曲线')
    ax.set_xlim(0, 24)
    ax.set_xticks(range(0, 25, 2))
    ax.legend(loc='upper right')
    ax.grid(True, alpha=0.3)

    plt.tight_layout()
    return fig
```

### 7.2 多电价方案支持

```python
# 🔲 电价方案管理（预留）
class PriceSchemeManager:
    """电价方案管理器"""

    _schemes: Dict[str, ElectricityPriceParams] = {}
    _current: str = "default"

    @classmethod
    def register(cls, name: str, scheme: ElectricityPriceParams):
        """注册电价方案"""
        cls._schemes[name] = scheme

    @classmethod
    def switch(cls, name: str) -> bool:
        """切换电价方案"""
        if name in cls._schemes:
            cls._current = name
            return True
        return False

    @classmethod
    def get_current(cls) -> ElectricityPriceParams:
        """获取当前电价方案"""
        return cls._schemes.get(cls._current)
```
