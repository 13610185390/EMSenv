## 1. 模块概述

### 1.1 功能描述
故障诊断模块是面向售后工程师的核心培训模块，提供故障模拟、报警分析、排查指南和案例复盘功能，帮助学员掌握储能系统常见故障的诊断和处理能力。

### 1.2 模块职责
- 模拟各类储能系统故障
- 展示报警触发和处理逻辑
- 提供交互式故障排查指南
- 管理故障案例库
- 记录排查练习过程和结果
- 评估故障处理能力

### 1.3 依赖关系
```
fault_diagnosis.py
    ├── config/fault_scenarios.py   # 故障场景配置
    ├── models/fault.py             # 故障相关数据模型
    ├── modules/simulation.py       # 仿真引擎（注入故障）
    └── modules/learning_center.py  # 关联知识点
```

```python
from dataclasses import dataclass, field
from typing import Optional, List, Dict, Any, Callable
from enum import Enum
from datetime import datetime


class FaultCategory(Enum):
    """故障类别"""
    BATTERY = "电池类"
    PCS = "变流器类"
    COMMUNICATION = "通信类"
    PROTECTION = "保护类"
    CONTROL = "控制类"
    ENVIRONMENT = "环境类"
    # 🔲 扩展预留
    EXT_CATEGORY_1 = "_扩展类别1"


class FaultSeverity(Enum):
    """故障严重程度"""
    INFO = "提示"       # 蓝色，仅记录
    WARNING = "一般"    # 黄色，提醒
    ALARM = "严重"      # 橙色，降功率
    CRITICAL = "紧急"   # 红色，停机


class FaultStatus(Enum):
    """故障状态"""
    ACTIVE = "活动"
    ACKNOWLEDGED = "已确认"
    RESOLVED = "已解决"
    CLEARED = "已清除"


@dataclass
class FaultType:
    """故障类型定义"""

    id: str                             # 故障代码，如 F001
    name: str                           # 故障名称
    category: FaultCategory
    severity: FaultSeverity

    # 描述
    description: str = ""               # 故障描述
    possible_causes: List[str] = field(default_factory=list)  # 可能原因
    symptoms: List[str] = field(default_factory=list)          # 故障现象

    # 处理
    immediate_action: str = ""          # 立即动作
    recovery_condition: str = ""        # 恢复条件
    handling_steps: List[str] = field(default_factory=list)   # 处理步骤

    # 教学
    learning_objectives: List[str] = field(default_factory=list)
    related_knowledge: List[str] = field(default_factory=list)
    difficulty: str = "intermediate"    # beginner, intermediate, advanced

    # 🔲 扩展预留
    ext_data: Dict[str, Any] = field(default_factory=dict)
```

```python
    related_knowledge: List[str] = field(default_factory=list)


@dataclass
class FaultCaseLibrary:
    """故障案例库"""

    cases: Dict[str, FaultCase] = field(default_factory=dict)

    def get_cases_by_type(self, fault_type_id: str) -> List[FaultCase]:
        return [c for c in self.cases.values() if c.fault_type_id == fault_type_id]

    def search_cases(self, keyword: str) -> List[FaultCase]:
        keyword_lower = keyword.lower()
        return [
            c for c in self.cases.values()
            if keyword_lower in c.title.lower() or
               keyword_lower in c.phenomenon.lower() or
               keyword_lower in c.root_cause.lower()
        ]
```

### 4.2 故障注入逻辑

```python
class FaultSimulator:
    """故障模拟器"""

    def __init__(self):
        self._active_faults: Dict[str, FaultInstance] = {}
        self._alarm_records: List[AlarmRecord] = []
        self._fault_types = _init_fault_types()

    def inject_fault(
        self,
        fault_type_id: str,
        fault_params: Optional[Dict[str, Any]] = None
    ) -> FaultInstance:
        """注入故障"""

        fault_type = self._fault_types.get(fault_type_id)
        if not fault_type:
            raise ValueError(f"未知的故障类型: {fault_type_id}")

        # 创建故障实例
        instance_id = f"{fault_type_id}_{datetime.now().strftime('%Y%m%d%H%M%S')}"
        instance = FaultInstance(
            id=instance_id,
            fault_type_id=fault_type_id,
            triggered_at=datetime.now(),
            fault_data=fault_params or {}
        )

        # 根据故障类型执行具体的模拟
        if fault_type_id == "F001":
            self._inject_soc_jump(instance, fault_params)
        elif fault_type_id == "F002":
            self._inject_comm_fault(instance, fault_params)
        elif fault_type_id == "F003":
            self._inject_power_limit(instance, fault_params)
        elif fault_type_id == "F004":
            self._inject_over_temp(instance, fault_params)

        # 生成报警
        self._generate_alarm(fault_type, instance)

        self._active_faults[instance_id] = instance
        return instance

    def _inject_soc_jump(
        self,
        instance: FaultInstance,
        params: Optional[Dict[str, Any]]
    ):
        """注入SOC跳变故障"""
        # 默认参数
        jump_value = params.get("jump_value", -15) if params else -15

        # 获取当前SOC
        current_soc = self._get_current_soc()  # 从仿真状态获取

        # 模拟SOC跳变
        new_soc = max(0, min(100, current_soc + jump_value))

        instance.fault_data = {
            "soc_before": current_soc,
            "soc_after": new_soc,
            "jump_value": jump_value,
            "timestamp": datetime.now().isoformat()
        }

        # 触发到仿真模块
        self._trigger_soc_change(new_soc)

    def _inject_comm_fault(
        self,
        instance: FaultInstance,
        params: Optional[Dict[str, Any]]
    ):
        """注入通信中断故障"""
        timeout_seconds = params.get("timeout_seconds", 10) if params else 10

        instance.fault_data = {
            "comm_target": "BMS",
            "timeout_seconds": timeout_seconds,
            "last_data_time": datetime.now().isoformat()
        }

        # 模拟通信中断
        self._trigger_comm_timeout("BMS")

    def _generate_alarm(self, fault_type: FaultType, instance: FaultInstance):
        """生成报警记录"""
        alarm = AlarmRecord(
            id=f"ALM_{instance.id}",
            alarm_code=fault_type.id,
            alarm_name=fault_type.name,
            severity=fault_type.severity,
            triggered_at=datetime.now(),
            trigger_value=instance.fault_data,
            component=fault_type.category.value
        )
        self._alarm_records.append(alarm)

    def clear_fault(self, instance_id: str) -> bool:
        """清除故障"""
        if instance_id not in self._active_faults:
            return False

        instance = self._active_faults[instance_id]
        instance.status = FaultStatus.CLEARED
        instance.resolved_at = datetime.now()

        # 恢复系统状态
        fault_type_id = instance.fault_type_id
        if fault_type_id == "F001":
            self._restore_soc()
        elif fault_type_id == "F002":
            self._restore_comm()

        del self._active_faults[instance_id]
        return True
```

### 4.4 诊断练习评估

```python
class DiagnosisPracticeManager:
    """诊断练习管理器"""

    def __init__(self):
        self._practices: Dict[str, DiagnosisPractice] = {}
        self._guides = _init_diagnosis_guides()

    def start_practice(
        self,
        user_id: str,
        fault_type_id: str
    ) -> DiagnosisPractice:
        """开始诊断练习"""

        guide_id = f"DG_{fault_type_id}"
        guide = self._guides.get(guide_id)
        if not guide:
            raise ValueError(f"未找到故障{fault_type_id}的诊断指南")

        practice = DiagnosisPractice(
            id=f"PRC_{user_id}_{datetime.now().strftime('%Y%m%d%H%M%S')}",
            user_id=user_id,
            fault_type_id=fault_type_id,
            guide_id=guide_id,
            started_at=datetime.now()
        )

        self._practices[practice.id] = practice
        return practice

    def submit_step(
        self,
        practice_id: str,
        step_number: int,
        answers: Dict[str, Any]
    ) -> Dict[str, Any]:
        """提交步骤答案"""

        practice = self._practices.get(practice_id)
        if not practice:
            return {"error": "练习不存在"}

        guide = self._guides.get(practice.guide_id)
        step = guide.steps[step_number - 1]

        # 记录答案
        practice.answers[str(step_number)] = answers
        practice.steps_completed.append(step_number)

        # 检查答案
        result = {
            "step_number": step_number,
            "feedback": "",
            "next_step": None
        }

        # 判断是否有决策分支
        for check_item in step.check_items:
            if "correct" in check_item:
                item_id = check_item["id"]
                if item_id in answers:
                    user_answer = answers[item_id]
                    if user_answer != check_item["correct"]:
                        # 答错，给提示
                        result["feedback"] = f"请再考虑一下：{check_item['question']}"
                        result["hint"] = step.hint
                        if "next_if_no" in check_item:
                            result["branch"] = check_item["next_if_no"]

        # 下一步
        if step_number < len(guide.steps):
            result["next_step"] = step_number + 1
        else:
            result["is_final"] = True

        return result

    def complete_practice(
        self,
        practice_id: str,
        conclusion: str
    ) -> Dict[str, Any]:
        """完成练习"""

        practice = self._practices.get(practice_id)
        if not practice:
            return {"error": "练习不存在"}

        guide = self._guides.get(practice.guide_id)

        # 评估结论
        is_correct = False
        correct_conclusion = ""
        for c in guide.possible_conclusions:
            if c["id"] == conclusion or c["conclusion"] == conclusion:
                is_correct = True
                correct_conclusion = c["conclusion"]
                break

        # 计算得分
        score = self._calculate_score(practice, is_correct)

        # 更新练习记录
        practice.completed_at = datetime.now()
        practice.time_spent_seconds = int(
            (practice.completed_at - practice.started_at).total_seconds()
        )
        practice.final_conclusion = conclusion
        practice.is_correct = is_correct
        practice.score = score

        # 生成反馈
        feedback = self._generate_feedback(practice, is_correct, guide)

        return {
            "is_correct": is_correct,
            "score": score,
            "feedback": feedback,
            "correct_conclusion": correct_conclusion,
            "time_spent": practice.time_spent_seconds
        }

    def _calculate_score(
        self,
        practice: DiagnosisPractice,
        is_correct: bool
    ) -> float:
        """计算得分"""
        base_score = 60 if is_correct else 30

        # 根据使用提示次数扣分
        hint_penalty = practice.hints_used * 5

        # 根据时间给分（合理时间内完成加分）
        guide = self._guides.get(practice.guide_id)
        expected_seconds = guide.expected_time_minutes * 60
        actual_seconds = practice.time_spent_seconds

        if actual_seconds <= expected_seconds:
            time_bonus = 20
        elif actual_seconds <= expected_seconds * 1.5:
            time_bonus = 10
        else:
            time_bonus = 0

        score = base_score - hint_penalty + time_bonus
        return max(0, min(100, score))

    def _generate_feedback(
        self,
        practice: DiagnosisPractice,
        is_correct: bool,
        guide: DiagnosisGuide
    ) -> str:
        """生成反馈"""
        if is_correct:
            feedback = "诊断正确！"
        else:
            feedback = "诊断结论不正确。"

        feedback += f"\n\n耗时: {practice.time_spent_seconds // 60}分{practice.time_spent_seconds % 60}秒"
        feedback += f"\n使用提示: {practice.hints_used}次"

        if not is_correct:
            feedback += "\n\n建议复习以下知识点："
            fault_type = _init_fault_types().get(practice.fault_type_id)
            if fault_type:
                for kp in fault_type.related_knowledge:
                    feedback += f"\n- 知识点 {kp}"

        return feedback
```
